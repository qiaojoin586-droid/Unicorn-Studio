CREATE TABLE sys_user (
  id                     BIGINT PRIMARY KEY AUTO_INCREMENT,
  username               VARCHAR(50) NOT NULL UNIQUE,
  password               VARCHAR(100) NOT NULL COMMENT 'BCrypt',
  nickname               VARCHAR(50),
  avatar                 VARCHAR(255),

CREATE TABLE sys_role (
  id          BIGINT PRIMARY KEY AUTO_INCREMENT,
  code        VARCHAR(50) NOT NULL UNIQUE COMMENT 'SUPER_ADMIN/EDITOR/OPERATOR/AUDITOR',
  name        VARCHAR(50) NOT NULL,
  description VARCHAR(255),
  builtin     TINYINT NOT NULL DEFAULT 0 COMMENT '1=内置不可删改',
  status      TINYINT NOT NULL DEFAULT 1,
  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE sys_permission (
  id         BIGINT PRIMARY KEY AUTO_INCREMENT,
  code       VARCHAR(50) NOT NULL UNIQUE,
  name       VARCHAR(50) NOT NULL,
  module     VARCHAR(50),
  type       VARCHAR(20) NOT NULL DEFAULT 'BUTTON' COMMENT 'MENU/BUTTON',
  sort_order INT NOT NULL DEFAULT 0

CREATE TABLE sys_user_role (
  user_id BIGINT NOT NULL,
  role_id BIGINT NOT NULL,
  PRIMARY KEY (user_id, role_id),

  CONSTRAINT fk_ur_role FOREIGN KEY (role_id) REFERENCES sys_role(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE sys_role_permission (
  role_id       BIGINT NOT NULL,
  permission_id BIGINT NOT NULL,
  PRIMARY KEY (role_id, permission_id),

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE site_settings (

  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP

CREATE TABLE sys_operation_log (

  created_by BIGINT,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP

## 3. 后台 API 全量清单

## 5. 关键实现要点（代码级）

### 5.1 双过滤器链（后端）

```java
@Configuration
@EnableWebSecurity
@EnableMethodSecurity   // 开启 @PreAuthorize
public class AdminSecurityConfig {

    @Bean
    @Order(1)  // 后台管理链：/api/admin/**
    public SecurityFilterChain adminChain(HttpSecurity http, JwtAuthFilter jwtFilter) throws Exception {
        http.securityMatcher("/api/admin/**")
            .csrf(csrf -> csrf.disable())
            .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(auth -> auth.anyRequest().authenticated())
            .addFilterBefore(jwtFilter, UsernamePasswordAuthenticationFilter.class)
            .exceptionHandling(e -> e
                .authenticationEntryPoint((req, res, ex) ->
                    writeJson(res, 401, 40100, "未认证或登录已过期"))
                .accessDeniedHandler((req, res, ex) ->
                    writeJson(res, 403, 40300, "无操作权限")));
        return http.build();
    }

    @Bean
    @Order(2)  // 公开链：/api/**（含 stats/pv、前台注册等，各接口自带限流）
    public SecurityFilterChain publicChain(HttpSecurity http) throws Exception {
        http.securityMatcher("/api/**")
            .csrf(csrf -> csrf.disable())
            .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(auth -> auth.anyRequest().permitAll());
        return http.build();
    }
}
```

要点：
- 401/403 统一返回 `ApiResponse` 信封（`{code:40100, message, data:null}`），前端拦截器据此触发刷新或跳登录；
- JWT 过滤器从 `Authorization: Bearer` 解析 → 查库加载角色/权限（**实时**，角色修改即时生效）→ 注入 `UsernamePasswordAuthenticationToken(principal, null, authorities)`；
- `SUPER_ADMIN` 角色代码级绕过：`@PreAuthorize` 判定逻辑为 `hasRole('SUPER_ADMIN') or hasAuthority('xxx')`——用自定义 `PermissionEvaluator` 或在加载 authorities 时对该角色注入全部权限码（**推荐后者**，注解写法不变：加载时若含 SUPER_ADMIN 则 authorities = 全量权限码集合）。
- 链评估顺序：`@Order(1)` admin 链优先匹配 `/api/admin/**`，命中后不再进入 `@Order(2)` 公开链；其余 `/api/*` 请求落入公开链（permitAll + 各接口限流）。securityMatcher 允许重叠，顺序由 `@Order` 决定。
- CSRF：两条链均 `csrf.disable()`（STATELESS + Bearer 的标准做法）。公开写接口 `/api/comments`、`/api/messages` 无 token，需在 publicChain 增加 Origin/Referer 校验过滤器，叠加限流 + 先审后发兜底。

    private static final Safelist SAFELIST = new Safelist(Safelist.relaxed()
        .addTags("img", "h1", "h2", "h3", "h4", "table", "thead", "tbody", "tr", "td", "th", "code", "pre", "span")

@Aspect @Component
public class AuditLogAspect {
    @Around("@annotation(auditLog)")
    public Object around(ProceedingJoinPoint pjp, AuditLog auditLog) throws Throwable {
        long start = System.currentTimeMillis();
        try {
            Object result = pjp.proceed();
            save(pjp, auditLog, "success", null, System.currentTimeMillis() - start);
            return result;
        } catch (Throwable t) {
            save(pjp, auditLog, "failed", t.getMessage(), System.currentTimeMillis() - start);
            throw t;
        }
    }
    // save: 参数 JSON 序列化后脱敏（password/token/refreshToken → "***"）、IP、耗时

## 8. 联调与部署要点

1. **开发环境**：前台 Vite 5173 → 代理 8080；admin Vite 5174 → 代理 8080；两者共享同一后端，无需额外 mock。
2. **生产 nginx**（在升级方案部署模板上追加）：

```nginx
location /admin/ {
    alias /srv/unicorn/admin/;          # admin 构建产物（独立 dist）
    try_files $uri $uri/ /admin/index.html;
}
location /uploads/ {
    alias /srv/unicorn/data/uploads/;
    expires 1y; add_header Cache-Control "public, immutable";
}
# /api/admin/** 与 /api/** 统一走此反代（SSE 必需：关缓冲 + 长读超时）
location /api/ {
    proxy_pass http://127.0.0.1:8080;
    proxy_buffering off;             # SSE 必需，否则流式响应退化为一次性返回
    proxy_read_timeout 300s;         # SSE 长连接
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
}
```

3. **环境变量清单**（后端新增）：`ADMIN_JWT_SECRET`（≥32 字符）、`ADMIN_INIT_PASSWORD`（仅首次启动使用）、`ADMIN_JWT_REFRESH_EXPIRE_DAYS`（默认 7）、`FILE_UPLOAD_MAX_SIZE`（默认 10MB）、`BACKUP_RETENTION_COUNT`（默认 7）、`RATE_LIMIT_*`（登录/评论/留言/埋点阈值，默认见 §3.7）、`MYSQL_*`（升级方案 4.2）、`SMTP_*`（邮件通知可选）、`AI_API_KEY`（已有）。
4. **数据迁移**：H2 → MySQL 时现有 4 条 learning_resource 数据以 `V6__seed_learning_resource.sql`（Flyway 追加脚本，内容为从 H2 导出的 INSERT）迁入，保证全量执行后数据完整；后台补录仅作兜底。**不再使用 data.sql 全量清空机制**（升级方案 P1-1）。
5. **outDir 衔接**：升级方案 2.4 执行后前台 outDir 为 `dist/`；admin 前端独立构建（`outDir: 'admin/dist'` + `base: '/admin/'`），两者互不影响，nginx 路径见上文。
